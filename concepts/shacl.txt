```Example from the web:
ex:PersonShape
a sh:NodeShape ;
sh:targetClass ex:Person ;    # Applies to all persons
sh:property [                 # _:b1
	sh:path ex:ssn ;           # constrains the values of ex:ssn
	sh:maxCount 1 ;
	sh:datatype xsd:string ;
	sh:pattern &quot;^\\d{3}-\\d{2}-\\d{4}$&quot; ;
] ;
sh:property [                 # _:b2
	sh:path ex:worksFor ;
	sh:class ex:Company ;
	sh:nodeKind sh:IRI ;
] ;
sh:closed true ;
sh:ignoredProperties ( rdf:type ) .
```
