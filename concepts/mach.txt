Mach is a small, explicit, self-hosted systems programming language for compilers, kernels, runtimes and games, with no hidden control flow, no hidden allocation and no type inference.

```Example from the web:
use std.print;
use std.runtime;

fun fibonacci(n: u64) u64 {
    var a: u64 = 0;
    var b: u64 = 1;
    var i: u64 = 0;
    for (i &lt; n) {
        val t: u64 = b;
        b = a + b;
        a = t;
        i = i + 1;
    }
    ret a;
}

#[symbol(&quot;main&quot;)]
fun main(argc: i64, argv: **u8) i64 {
    print.printlnf(&quot;fib(10) = {}&quot;, fibonacci(10));
    ret 0;
}
```
