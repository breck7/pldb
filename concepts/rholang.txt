new helloworld, stdout(`rho:io:stdout`) in {
   contract helloworld( world ) = {
       for( @msg &lt;- world ) {
           stdout!(msg)
       }
   } |
   new world, world2 in {
        helloworld!(*world) |
        world!(&quot;Hello World&quot;) |
         helloworld!(*world2) |
