```Example from the web:
# simple program that iteratively calculates factorials

fn range(a, b) {
  tmp := []
  for i := a; i &lt; b; i += 1 {
    tmp-&gt;push(i)
  }

  return tmp
}

fn fact(n) {
  tmp := 1
  while n &gt; 0 {
    tmp *= n
    n -= 1
  }

  return tmp
}

for i &lt;- range(0, 7) {
  echo &quot;fact(#{i}): #{fact(i)}&quot;
}
```
