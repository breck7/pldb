```Example from the web:
package main

import &quot;core:fmt&quot;

main :: proc() {
 program := &quot;+ + * 😃 - /&quot;;
 accumulator := 0;

 for token in program {
   switch token {
   case &#39;+&#39;: accumulator += 1;
   case &#39;-&#39;: accumulator -= 1;
   case &#39;*&#39;: accumulator *= 2;
   case &#39;/&#39;: accumulator /= 2;
   case &#39;😃&#39;: accumulator *= accumulator;
   case: // Ignore everything else
   }
 }

 fmt.printf(&quot;The program \&quot;%s\&quot; calculates the value %d\n&quot;,
            program, accumulator);
}
```
