use io

runProg: &#39;{IO, Exception} a -&gt; &#39;{IO} ()
runProg f = &#39;let
  printErr err = match err with
    Failure _ errMsg _ -&gt; handle putBytes (stdHandle StdErr) (toUtf8 errMsg) with cases
      {raise _ -&gt; _} -&gt; ()
      {_} -&gt; ()
  match catch f with
    Left err -&gt; printErr err
    Right _ -&gt; ()

main: &#39;{IO} ()
main = runProg &#39;let
  printLine &quot;Hello, world!&quot;
