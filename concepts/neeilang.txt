fn main() : Int {
  var a : Int = 5;
  var b : Int = 3;

  print a + b;    // 8
  print a - b;    // 2
  print b - a;    // -2

  if (a &gt;= 5) {
    print &quot;a &gt;= 5&quot;; // a &gt;= 5
  }

  if (a &gt; 5) {
    print &quot;a &gt; 5&quot;;  // Not executed
  }

  if (a &lt; 5) {
    print &quot;a &lt; 5&quot;;  // Not executed
  }

  if (5 &lt;= a) {
    print &quot;5 &lt;= a&quot;;  // 5 &lt;= a
  }
